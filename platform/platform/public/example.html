import os
import pandas as pd
import numpy as np
import torch
from sentence_transformers import SentenceTransformer

### Device
device = "cuda" if torch.cuda.is_available() else "cpu"

### Load Model (Must be available in the script's directory)
model = SentenceTransformer("./checkpoint", local_files_only=True)

### Load Keywords
keywords = []

with open("./keywords.txt", "r") as f:
    i = 0
    for line in f:
        keywords.append(line.strip())
        i += 1
    print(f"There is {i} keywords.")
    
### Load Abstracts
abstracts = pd.read_csv("./articles.csv")["abstract"].tolist()
print(f"There is {len(abstracts)} articles.")

### Embeddings
K = torch.from_numpy(model.encode(
    sentences=keywords,
    batch_size=64,
    show_progress_bar=True,
    normalize_embeddings=True,
    device=device
))

A = torch.from_numpy(model.encode(
    sentences=abstracts,
    batch_size=64,
    show_progress_bar=True,
    normalize_embeddings=True,
    device=device
))

### Compute Scores
scores = torch.sigmoid(A @ K.T).numpy()

### Save Scores
np.save("./scores.npy", scores, allow_pickle=False)
